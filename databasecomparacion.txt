root@debian:/var/www/postgres# cat > /tmp/comparar_registros.sh << 'EOF'
#!/bin/bash

echo "=== Obteniendo registros del servidor 200 ==="
PGPASSWORD=09dgc06mcc psql -h 192.168.10.200 -U postgres -d bdSueldos -t -A -c "
SELECT tablename, 
       (xpath('/row/cnt/text()', xml_count))[1]::text::bigint as count
FROM (
    SELECT tablename, 
           query_to_xml('SELECT COUNT(*) as cnt FROM public.' || quote_ident(tablename), false, true, '') as xml_count
    FROM pg_tables 
    WHERE schemaname = 'public'
    ORDER BY tablename
) t;" > /tmp/registros_200.txt

echo "=== Obteniendo registros de la VPS ==="
docker exec dbSueldos psql -U postgres -d bdsueldos -t -A -c "
SELECT tablename, 
       (xpath('/row/cnt/text()', xml_count))[1]::text::bigint as count
FROM (
    SELECT tablename, 
           query_to_xml('SELECT COUNT(*) as cnt FROM public.' || quote_ident(tablename), false, true, '') as xml_count
    FROM pg_tables 
    WHERE schemaname = 'public'
    ORDER BY tablename
) t;" > /tmp/registros_vps.txt

echo ""
echo "=== COMPARACIÓN ==="
echo "Formato: TABLA | 200 | VPS | DIFERENCIA"
echo "----------------------------------------"

# Combinar y comparar
join -t'|' -a1 -a2 <(sort /tmp/registros_200.txt) <(sort /tmp/registros_vps.txt) | while IFS='|' read tabla cnt200 cntvps; do
    cnt200=${cnt200:-0}
    cntvps=${cntvps:-0}
    diff=$((cntvps - cnt200))
    if [ "$diff" -ne 0 ]; then
        echo "❌ $tabla | $cnt200 | $cntvps | $diff"
    else
        echo "✅ $tabla | $cnt200 | $cntvps | OK"
    fi
done

echo ""
echo "=== RESUMEN ==="
echo "Tablas en 200: $(wc -l < /tmp/registros_200.txt)"
echo "Tablas en VPS: $(wc -l < /tmp/registros_vps.txt)"
EOF

chmod +x /tmp/comparar_registros.sh
